import java.io.*;

public class CodigoConFallos {

    // Caso 1: saluda al nombre que escribe el usuario
    static void saludar(String nombre) throws Exception {
        String comando = "echo Hola " + nombre;
        Process p = new ProcessBuilder("sh", "-c", comando).start();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
    }

    // Caso 2: cuenta cuántos números genera "seq"
    static void contarNumeros() throws Exception {
        Process p = new ProcessBuilder("seq", "1", "1000000").start();
        p.waitFor();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        int n = 0;
        while (br.readLine() != null) n++;
        System.out.println("Números generados: " + n);
    }

    // Caso 3: ordena unas palabras con el comando "sort"
    static void ordenar() throws Exception {
        Process p = new ProcessBuilder("sort").start();
        p.getOutputStream().write("pera\nmanzana\nbanana\n".getBytes());
        p.getOutputStream().flush();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
    }

    // Caso 4: muestra el contenido de un fichero
    static void mostrar(String fichero) throws Exception {
        Process p = new ProcessBuilder("cat", fichero).start();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
        p.waitFor();
        System.out.println("Fichero mostrado correctamente");
    }

    public static void main(String[] args) throws Exception {
        String texto = args.length > 1 ? args[1] : "Ana";
        switch (args[0]) {
            case "1" -> saludar(texto);
            case "2" -> contarNumeros();
            case "3" -> ordenar();
            case "4" -> mostrar(texto);
        }
    }
}



Errores vistos por Claude con la skill:
ERRORES ENCONTRADOS EN CodigoConFallos.java

CASO 1 - saludar()
Error: Inyección de comandos.
Causa: se usa "sh -c" con el nombre concatenado, así que el shell interpreta lo que escriba el usuario (por ejemplo "Ana; rm -rf ~" ejecutaría también el rm).
Solución: pasar argumentos separados: new ProcessBuilder("echo", "Hola", nombre).

CASO 2 - contarNumeros()
Error: waitFor() antes de leer la salida (bloqueo).
Causa: seq genera unos 7 MB, más que el buffer de la tubería. seq espera a que alguien lea y Java espera a que seq termine, así que el programa se cuelga.
Solución: leer la salida primero y llamar a waitFor() después del bucle.

CASO 3 - ordenar()
Error: no se cierra getOutputStream() (bloqueo).
Causa: sort no emite nada hasta recibir fin de entrada (EOF). Como stdin nunca se cierra, sort espera datos y Java espera la salida.
Solución: cambiar flush() por close(), que además vacía el buffer.

CASO 4 - mostrar()
Error: se ignora el código de salida.
Causa: si el fichero no existe, cat termina con código 1 y aun así se imprime "Fichero mostrado correctamente". Además, un nombre que empiece por "-" se interpretaría como opción.
Solución: comprobar que waitFor() == 0 antes de dar el mensaje de éxito, y usar "--" antes del nombre: new ProcessBuilder("cat", "--", fichero).

MAIN (fallo menor, no corregido)
Error: sin argumentos, args[0] lanza ArrayIndexOutOfBoundsException.

import java.io.*;

public class CodigoConFallos {

    // Caso 1: saluda al nombre que escribe el usuario
    static void saludar(String nombre) throws Exception {
        Process p = new ProcessBuilder("echo", "Hola", nombre).start();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
    }

    // Caso 2: cuenta cuántos números genera "seq"
    static void contarNumeros() throws Exception {
        Process p = new ProcessBuilder("seq", "1", "1000000").start();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        int n = 0;
        while (br.readLine() != null) n++;
        p.waitFor();
        System.out.println("Números generados: " + n);
    }

    // Caso 3: ordena unas palabras con el comando "sort"
    static void ordenar() throws Exception {
        Process p = new ProcessBuilder("sort").start();
        p.getOutputStream().write("pera\nmanzana\nbanana\n".getBytes());
        p.getOutputStream().close();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
    }

    // Caso 4: muestra el contenido de un fichero
    static void mostrar(String fichero) throws Exception {
        Process p = new ProcessBuilder("cat", "--", fichero).start();
        BufferedReader br = new BufferedReader(new InputStreamReader(p.getInputStream()));
        String linea;
        while ((linea = br.readLine()) != null) System.out.println(linea);
        if (p.waitFor() == 0) System.out.println("Fichero mostrado correctamente");
        else System.out.println("Error al mostrar el fichero");
    }

    public static void main(String[] args) throws Exception {
        String texto = args.length > 1 ? args[1] : "Ana";
        switch (args[0]) {
            case "1" -> saludar(texto);
            case "2" -> contarNumeros();
            case "3" -> ordenar();
            case "4" -> mostrar(texto);
        }
    }
